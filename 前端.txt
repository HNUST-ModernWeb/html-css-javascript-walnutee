  </div>
</template>

<script setup>
import { ref } from 'vue'
import axios from 'axios'
import { useRouter } from 'vue-router'
const router = useRouter()
const username = ref('')
const password = ref('')

const login = async () => {
  const res = await axios.post('http://localhost:8080/user/login', {
    username: username.value,
    password: password.value
  })
  if (res.data.code === 200) {
    localStorage.setItem('user', username.value)
    alert('登录成功')
    router.push('/home')
  }
}
</script>

<template>
  <div class="home">
    <div class="card" v-for="item in list" :key="item.id">
      <h3>{{ item.title }}</h3>
      <p>作者：{{ item.username }}</p>
      <p>{{ item.content }}</p>
      <img v-if="item.img_url" :src="item.img_url" style="max-width:100%" />

      <div class="like">
        <button @click="like(item.id)">👍 点赞 {{ item.like_count }}</button>
      </div>

      <div class="comment">
        <input v-model="cContent" placeholder="写评论..." />
        <button @click="sendComment(item.id)">发送</button>
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue'
import axios from 'axios'
const list = ref([])
const cContent = ref('')

const getList = async () => {
  const res = await axios.get('http://localhost:8080/post/list')
  list.value = res.data.data
}

const like = async (id) => {
  await axios.post(`http://localhost:8080/post/like/${id}`)
  getList()
}

const sendComment = async (postId) => {
  await axios.post('http://localhost:8080/comment/add', {
    postId, content: cContent.value
  })
  cContent.value = ''
  getList()
}

onMounted(() => getList())
</script>

<template>
  <div class="publish">
    <h2>发布动态</h2>
    <input v-model="title" placeholder="标题" />
    <textarea v-model="content" placeholder="内容"></textarea>
    <input type="file" @change="upload" />
    <button @click="submit">发布</button>
  </div>
</template>

<script setup>
import { ref } from 'vue'
import axios from 'axios'
import { useRouter } from 'vue-router'
const router = useRouter()
const title = ref('')
const content = ref('')
const imgUrl = ref('')

const upload = async (e) => {
  const fd = new FormData()
  fd.append('file', e.target.files[0])
  const res = await axios.post('http://localhost:8080/upload', fd)
  imgUrl.value = res.data.data
}

const submit = async () => {
  await axios.post('http://localhost:8080/post/add', {
    title: title.value,
    content: content.value,
    imgUrl: imgUrl.value,
    userId: 1
  })
  alert('发布成功')
  router.push('/home')
}
</script>
