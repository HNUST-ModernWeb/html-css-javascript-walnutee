CREATE DATABASE IF NOT EXISTS social;
USE social;

CREATE TABLE user (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(20),
    password VARCHAR(20)
);

CREATE TABLE post (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(50),
    content TEXT,
    img_url VARCHAR(255),
    user_id INT,
    like_count INT DEFAULT 0
);

CREATE TABLE comment (
    id INT PRIMARY KEY AUTO_INCREMENT,
    post_id INT,
    content VARCHAR(255)
);
